cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_EXTENSIONS OFF)
project(spooler-monitor LANGUAGES CXX)

add_library(monitor_core STATIC monitor.cpp)
target_include_directories(monitor_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(monitor_core PUBLIC cxx_std_17)
set_target_properties(monitor_core PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
  find_library(CUPS_LIBRARY cups)
  if(NOT CUPS_LIBRARY)
    message(FATAL_ERROR "The macOS system CUPS library is required")
  endif()
  target_sources(monitor_core PRIVATE cups_backend.cpp)
  target_link_libraries(monitor_core PUBLIC "${CUPS_LIBRARY}")
elseif(WIN32)
  target_sources(monitor_core PRIVATE windows_backend.cpp)
  target_link_libraries(monitor_core PUBLIC winspool)
else()
  message(FATAL_ERROR "Only macOS and Windows backends are supported")
endif()

add_executable(spooler-monitor main.cpp)
target_link_libraries(spooler-monitor PRIVATE monitor_core)
if(MSVC)
  target_compile_options(monitor_core PUBLIC /utf-8 /W4)
elseif(MINGW)
  target_link_options(spooler-monitor PRIVATE -municode)
else()
  target_compile_options(monitor_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(monitor_fixture mock_main.cpp)
  target_link_libraries(monitor_fixture PRIVATE monitor_core)
  if(MINGW)
    target_link_options(monitor_fixture PRIVATE -municode)
  endif()
  add_executable(monitor_tests tests.cpp)
  target_link_libraries(monitor_tests PRIVATE monitor_core)
  add_test(NAME monitor_backend_protocol COMMAND monitor_tests)
  set_tests_properties(monitor_backend_protocol PROPERTIES TIMEOUT 20)
endif()
